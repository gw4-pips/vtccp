// WHAT IT DOES
//   After every scan the device calls onResult().  This script builds a
//   <DMCCResponse><DMSymVerResponse>…</DMSymVerResponse></DMCCResponse> XML
//   document and assigns it to output.content.  That content is then pushed
//   over the Network Client TCP connection to VTCCP (10.10.10.19:9004).
//   VTCCP's DmstListener detects the closing </DMCCResponse> tag, parses the
//   XML, maps every element to a VerificationRecord, and writes a fully-
//   populated XLSX row.
//
// TROUBLESHOOTING
//   If a column stays blank in VTCCP: the property name on the right-hand side
//   of the corresponding elem() call returned undefined.  Cognex exposes
//   slightly different property names across firmware revisions.  The property
//   names are annotated below; check the DataMan Scripting API Reference
//   (Settings → Help in DMST) for your exact firmware if a field is empty.
//
// COMPATIBILITY
//   ECMAScript 5 only — no const/let, no arrow functions, no template literals.

function onResult(decodeResults, readerProperties, outputResults) {

    // ── Helpers ───────────────────────────────────────────────────────────────

    // Safely stringify a value; returns "" for undefined/null.
    function s(v) {
        return (v === undefined || v === null) ? "" : String(v);
    }

    // Safely read a nested property path, e.g. prop(q, "lqz") → q.lqz ?? "".
    function prop(obj, key) {
        if (!obj) return "";
        var v = obj[key];
        return (v === undefined || v === null) ? "" : String(v);
    }

    // XML-safe string — replaces the five XML special characters.
    function esc(v) {
        return s(v)
            .replace(/&/g, "&amp;")
            .replace(/</g, "&lt;")
            .replace(/>/g, "&gt;")
            .replace(/"/g, "&quot;")
            .replace(/'/g, "&apos;");
    }

    // Emit one XML element.  Empty value → element present but empty.
    function elem(tag, val) {
        return "<" + tag + ">" + esc(val) + "</" + tag + ">\r\n";
    }

    // ISO 8601 timestamp from current wall-clock (device local time).
    function isoNow() {
        var t = new Date();
        function p(n) { return n < 10 ? "0" + n : String(n); }
        return t.getFullYear() + "-" + p(t.getMonth() + 1) + "-" + p(t.getDate())
             + "T" + p(t.getHours()) + ":" + p(t.getMinutes()) + ":" + p(t.getSeconds());
    }

    // ── Inputs ────────────────────────────────────────────────────────────────

    var r  = decodeResults[0];
    var rp = readerProperties;   // some firmware puts quality here instead of r

    // Null-safe property fetch — returns the value or null; never throws.
    // Using !== null/undefined (not truthy) so a quality object that happens
    // to stringify as 0 / false is still captured.
    function _pick(obj, key) {
        if (!obj) { return null; }
        var v = obj[key];
        return (typeof v !== "undefined" && v !== null) ? v : null;
    }

    // TrucheckMetric helpers (v1.10) — every sub-property on r.trucheck is a
    // TrucheckMetric with exactly two fields: .grade (letter) and .numericGrade.
    function tmGrade(tm) { return (tm && typeof tm["grade"]        !== "undefined") ? s(tm["grade"])        : ""; }
    function tmNum(tm)   { return (tm && typeof tm["numericGrade"] !== "undefined") ? s(tm["numericGrade"]) : ""; }

    // Metric helpers (v1.13) — Metric objects confirmed to have exactly:
    //   .grade  — letter string ("A"–"F") or "NA" (valid device output)
    //   .raw    — measurement number, or -1 (numeric sentinel = no value)
    // mmGrade() returns grade as-is — "NA" is a valid reported grade and
    //           is NOT suppressed (device genuinely reports it per scan).
    // mmVal()   returns raw as string; suppresses -1 (sentinel) → "".
    // mmPct()   returns raw×100, 1 dp; suppresses -1 → "" (-100% invalid).
    //           Use mmPct for percentage columns (SCPercent, ANUPercent…).
    function mmGrade(met) {
        if (!met) { return ""; }
        var _v = met["grade"];
        if (typeof _v === "undefined" || _v === null) { return ""; }
        return s(_v);  // "NA" is a valid device-returned grade — pass through
    }
    function mmVal(met) {
        if (!met) { return ""; }
        var _v = met["raw"];
        if (typeof _v === "undefined" || _v === null) { return ""; }
        return (_v === -1 || _v === "-1") ? "" : s(_v);
    }
    function mmPct(met) {
        if (!met) { return ""; }
        var _v = met["raw"];
        if (typeof _v === "undefined" || _v === null || _v === -1) { return ""; }
        var _n = parseFloat(_v);
        if (isNaN(_n)) { return ""; }
        // Raw is 0–1 ratio; convert to percentage with 1 decimal place.
        var _pct = Math.round(_n * 1000) / 10;
        return s(_pct);
    }
    // mmPctAuto (v1.16) — UEC.raw arrives already as 0–100 (giving 10000 with mmPct);
    // others (symbolContrast, ANU, GNU) arrive as 0–1 ratio.  Auto-detect: if the
    // raw value is > 1, assume it's already a percent; otherwise treat as ratio.
    function mmPctAuto(met) {
        if (!met) { return ""; }
        var _v = met["raw"];
        if (typeof _v === "undefined" || _v === null || _v === -1) { return ""; }
        var _n = parseFloat(_v);
        if (isNaN(_n)) { return ""; }
        var _pct = (_n > 1) ? _n : (_n * 100);
        return s(Math.round(_pct * 10) / 10);
    }

    // ── Quality-object discovery ──────────────────────────────────────────────
    // Try every known property path in priority order.  _qSource records which
    // one succeeded so <PushScriptDiag> can report it.  If every probe returns
    // null the <PushScriptDiag> element will say "none" and all grade columns
    // will be empty — paste the VS Output [VTCCP-DMST] RawXML line to the
    // VTCCP agent so it can identify the correct path.

    var q       = null;

        // ── v1.19 new grade emissions ─────────────────────────────────────────

        var _dd = _pick(q, "distributedDamageGrade");
        o += elem("DDGrade", (_dd && typeof _dd === "object") ? tmGrade(_dd) : s(_dd));
        //   averageGrade: separate TrucheckMetric carrying the ISO 15415 mean

        var _avg = _pick(q, "averageGrade");
        o += elem("AverageGrade",        tmGrade(_avg));
        o += elem("AverageGradeNumeric", mmVal(_avg));
        //   v1.20: customNote = device calibration metadata string

        o += elem("CustomNote",          s(_pick(q, "customNote")));

        o += elem("LLSGrade", tmGrade(_lls));

        o += elem("LQZGrade", tmGrade(_lqz));

        //   v1.19: TTR/RTR have their own q-side TrucheckMetrics (revealed by

        var _ttr = _pick(q, "topTransitionRatio");
        var _rtr = _pick(q, "rightTransitionRatio");

        o += elem("TTRGrade",   tmGrade(_ttr));
        o += elem("RTRPercent", mmPctAuto(_rtr));
        o += elem("RTRGrade",   tmGrade(_rtr));
        o += elem("TCTGrade",   tmGrade(_tct));
        o += elem("RCTGrade",   tmGrade(_rct));

        // ── Per-region parameters (≥ 32×32 / 2-region rectangular) ───────────
        //   v1.27 DebugQZObjs + DebugTTRCTObjs — CONFIRMED STRUCTURAL LIMIT.
        //   q.topQuietZone   = {grade:"X", numericGrade:8.8}  (2-key flat object)
        //   q.rightQuietZone = {grade:"X", numericGrade:8.8}  (same shape)
        //   q.topTransitionRatio  = {raw:-1, grade:"X", numericGrade:8.8}
        //   q.rightTransitionRatio = {raw:-1, grade:"X", numericGrade:8.8}
        //   q.topClockTrack  = {grade:"X", numericGrade:8.8}
        //   q.rightClockTrack = {grade:"X", numericGrade:8.8}
        //   These are firmware-internal compound objects.  "grade=X" and
        //   "numericGrade=8.8" are on an internal 0–10 firmware scale — NOT
        //   ISO letter grades.  The individual per-quadrant grades (ULQZ/URQZ/
        //   RUQZ/RLQZ/LQTTR/RQTTR/LQRTR/RQRTR/LQTCT/RQTCT/LQRCT/RQRCT) that
        //   appear as "A" in the DMST PDF are computed inside DMST from raw data
        //   and are NOT exposed through the push channel on firmware 6.1.16_sr4.
        //   These fields remain empty in the push XML.  This is the confirmed
        //   final answer — no further probing is warranted for this firmware.
        o += elem("ULQZGrade",        "");
        o += elem("URQZGrade",        "");
        o += elem("RUQZGrade",        "");
        o += elem("RLQZGrade",        "");
        o += elem("LLQZGrade",        "");
        o += elem("LRQZGrade",        "");
        o += elem("HClockTrackGrade", "");
        o += elem("VClockTrackGrade", "");
        o += elem("ULQTTRPercent", "");
        o += elem("ULQTTRGrade",   "");
        o += elem("URQTTRPercent", "");
        o += elem("URQTTRGrade",   "");
        o += elem("LLQTTRPercent", "");
        o += elem("LLQTTRGrade",   "");
        o += elem("LRQTTRPercent", "");
        o += elem("LRQTTRGrade",   "");
        o += elem("ULQRTRPercent", "");
        o += elem("ULQRTRGrade",   "");
        o += elem("URQRTRPercent", "");
        o += elem("URQRTRGrade",   "");
        o += elem("LLQRTRPercent", "");
        o += elem("LLQRTRGrade",   "");
        o += elem("LRQRTRPercent", "");
        o += elem("LRQRTRGrade",   "");
        o += elem("ULQTCTGrade",   "");
        o += elem("URQTCTGrade",   "");
        o += elem("LLQTCTGrade",   "");
        o += elem("LRQTCTGrade",   "");
        o += elem("ULQRCTGrade",   "");
        o += elem("URQRCTGrade",   "");
        o += elem("LLQRCTGrade",   "");
        o += elem("LRQRCTGrade",   "");

        // ── QR-specific grade parameters ──────────────────────────────────────
        //   v1.26 DebugTrucheckKeys confirmed: ALL QR grade params are top-level
        //   q keys (NOT in q.symbols[0], which is always null on this firmware).
        //   v1.30 BUG #3 FIX: guard all 8 params with _qrBranch.
        //   On non-QR symbologies (DM, Code 128, EAN-13) the underlying
        //   TrucheckMetric objects are populated by the firmware with a sentinel
        //   failure value ({grade:"F", numericGrade:0}).  This caused all 8 fields
        //   to emit "F" on DM live scans and Code 128 live scans (v1.29 catalog).
        //   For non-QR symbologies: emit "" (not applicable).
        //   For QR scans (_qrBranch=true): emit the TrucheckMetric grade as before.
        o += elem("ULPGrade", _qrBranch ? tmGrade(_pick(q, "upperLeftPattern"))      : "");
        o += elem("URPGrade", _qrBranch ? tmGrade(_pick(q, "upperRightPattern"))     : "");
        o += elem("LLPGrade", _qrBranch ? tmGrade(_pick(q, "lowerLeftPattern"))      : "");
        o += elem("HCTGrade", _qrBranch ? tmGrade(_pick(q, "horizontalClockTrack"))  : "");
        o += elem("VCTGrade", _qrBranch ? tmGrade(_pick(q, "verticalClockTrack"))    : "");
        o += elem("ALPGrade", _qrBranch ? tmGrade(_pick(q, "alignmentPatterns"))     : "");
        o += elem("VIBGrade", _qrBranch ? tmGrade(_pick(q, "versionInformationBlock")) : "");
        o += elem("FIBGrade", _qrBranch ? tmGrade(_pick(q, "formatInformationBlock")) : "");

        // ── 1D / ANSI summary ─────────────────────────────────────────────────
        var _ansiGrade = prop(m, "ansiGrade") || prop(m, "symbolAnsiGrade");
        o += elem("SymbolAnsiGrade", _ansiGrade);
        o += elem("AvgEdge",   prop(m, "avgEdge"));
        o += elem("AvgRlRd",   prop(m, "avgRlRd"));
        o += elem("AvgSC",     prop(m, "avgSc"));

        o += elem("AvgHQZ",    prop(m, "avgHqz"));

        // v1.24: BWG = Print Growth per comms-and-programming-guide §4. The
        // r.metrics.printGrowth is a Metric object {raw, grade}, not a scalar,
        // so prop() returns "[object Metric]" / undefined. Use mmPctAuto.
        o += elem("BWGPercent",    mmPctAuto(_pick(m, "printGrowth")));
        o += elem("Magnification", prop(m, "magnification"));
        o += elem("Ratio",         prop(m, "ratio"));
        o += elem("NominalXDim1D", prop(m, "nominalXDim1D") || prop(m, "nominalXDim1d"));

    // ── v1.24 + v1.25 wires (post-q-block) ───────────────────────────────────
    //   v1.25: Full JPEG image payload. v1.24 DebugJpegProbe confirmed:
    //   type=string, len≈9,912 for live DM / ≈22,260 for loaded QR — well
    //   within Network Client buffer. r.trucheck.jpegImage is a base64 JPEG
    //   string. Emit as-is; parser stores in VerificationRecord.JpegImageBase64.
    o += elem("JpegImageBase64", q ? s(_pick(q, "jpegImage")) : "");

    //   Promote r.trucheck.calibrationDate to first-class. v1.23 stuffed

    //   confirms r.trucheck.calibrationDate is a string timestamp directly
    //   addressable. CustomNote stays as-is (different data — user-set device
    //   note "Results from UPCE-44960 Cal. 23 JUN 2023").
    o += elem("CalibrationDate", q ? s(_pick(q, "calibrationDate")) : "");

    var _s3d = _pick(rp, "status3D");

    var _mpg = _pick(m, "minPassGrade");

    // -- v1.33 introspection probes -----------------------------------------
    //
    //   RETAINED (regression baselines):
    //     DebugMetricsKeys   -- r.metrics 30-key inventory (fw 6.1.16_sr4)
    //     DebugRSiblings     -- r-sibling inventory
    //     DebugRSymbology    -- r.symbology 9-key top-level baseline
    //     DebugImagePolarity -- q.general 7-key baseline
    //
    //   DROPPED (answered by v1.32 scan #11, 2026-05-24):
    //     DebugRImage -- r.image = camera hardware only; all 4 missing fields absent.
    //
    //   NEW PROBES:
    //     DebugRSymbologyNested -- sub-keys of r.symbology.size, .center, .corners[0]
    //     DebugQFIBKeys (QR)    -- ALL sub-keys of q.formatInformationBlock
    //     DebugQVIBKeys (QR)    -- ALL sub-keys of q.versionInformationBlock

    function _enumKV22(obj, label) {
        if (!obj) { return "(" + label + " null)"; }
        if (typeof obj !== "object") {
            return "(" + label + " " + (typeof obj) + "=" + String(obj).substring(0, 30) + ")";
        }
        var _out = "";
        for (var _k in obj) {
            var _v = obj[_k];
            var _t = typeof _v;
            var _vs;
            if (_t === "object" && _v !== null) {
                _vs = (typeof _v.length !== "undefined") ? ("[arr." + _v.length + "]") : "[obj]";
            } else {
                _vs = String(_v).substring(0, 40);
            }
            _out += _k + "=" + _vs + ";";
        }
        return _out || "(" + label + " empty)";
    }

    // RETAINED: regression baseline -- r.metrics key enumeration
    o += elem("DebugMetricsKeys", _enumKV22(m, "r.metrics"));

    // RETAINED: regression baseline -- r siblings
    var _v25known = { "trucheck": 1, "metrics": 1 };
    var _v25sibStr = "";
    if (r && typeof r === "object") {
        for (var _v25rk in r) {
            if (_v25known[_v25rk]) { continue; }
            var _v25rv = r[_v25rk];
            var _v25rvDesc;
            if (_v25rv === null) { _v25rvDesc = "null"; }
            else if (typeof _v25rv === "object") {
                _v25rvDesc = (typeof _v25rv.length !== "undefined") ? ("[arr." + _v25rv.length + "]") : "[obj]";
            } else {
                _v25rvDesc = (typeof _v25rv) + "=" + String(_v25rv).substring(0, 30);
            }
            _v25sibStr += _v25rk + "=" + _v25rvDesc + ";";
        }
    }
    o += elem("DebugRSiblings", _v25sibStr || "(no unknown r-siblings)");

    // RETAINED BASELINE: DebugRSymbology -- r.symbology 9-key TOP-LEVEL inventory.
    //   Confirmed DM + QR (live + IMAGE.LOAD): name/id/quality/contrast/moduleSize/
    //   corners/center/angle/size/failureCode. No ECLevel/dataMask/ECI/version key
    //   at top level. Nested objects probed separately below (DebugRSymbologyNested).
    o += elem("DebugRSymbology", _enumKV22(_sym, "r.symbology"));

    // NEW PROBE: DebugRSymbologyNested -- enumerate sub-keys of the three nested
    //   objects within r.symbology that _enumKV22 reported as "[obj]"/"[arr.4]".
    //   r.symbology.size   -- expected to encode symbol dimensions; may expose
    //                         QR version (size.version, size.modules, etc.)
    //   r.symbology.center -- center coordinate; expected {x, y} but unconfirmed
    //   r.symbology.corners[0] -- first of 4 corner coordinate objects
    var _symSize    = (_sym && _sym["size"])    ? _sym["size"]    : null;
    var _symCenter  = (_sym && _sym["center"])  ? _sym["center"]  : null;
    var _symCorners = (_sym && _sym["corners"]) ? _sym["corners"] : null;
    var _symC0      = (_symCorners && typeof _symCorners.length !== "undefined"
                       && _symCorners.length > 0) ? _symCorners[0] : null;
    o += elem("DebugRSymbologyNested",
        "size=" + _enumKV22(_symSize, "size")
        + "|center=" + _enumKV22(_symCenter, "center")
        + "|corners[0]=" + _enumKV22(_symC0, "corners[0]"));

    // NEW PROBE (QR only): DebugQFIBKeys -- enumerate ALL sub-keys of the
    //   q.formatInformationBlock TrucheckMetric object.
    //   RATIONALE: Prior probes read .grade only via tmGrade(). The QR Format
    //   Information Block encodes BOTH ECLevel (L/M/Q/H) AND DataMaskPattern (0-7)
    //   in the QR bitstream. If the firmware exposes these as additional sub-keys
    //   on this TrucheckMetric (beyond .grade/.numericGrade), they are readable here.
    //   This is the highest-probability unexplored path for both fields.
    //   Emits "(not QR)" on DM and 1D symbologies.
    var _fibDebug = _qrBranch
        ? _enumKV22(_pick(q, "formatInformationBlock"), "q.formatInformationBlock")
        : "(not QR)";
    o += elem("DebugQFIBKeys", _fibDebug);

    // NEW PROBE (QR only): DebugQVIBKeys -- enumerate ALL sub-keys of the
    //   q.versionInformationBlock TrucheckMetric object.
    //   RATIONALE: Prior probes read .grade only via tmGrade(). The VIB encodes
    //   the QR version number. A .version or .versionNumber sub-key would give us
    //   QR_Version directly without MatrixSize lookup.
    //   Note: VIBGrade="-" on v1/v3 QR (no VIB block); object may be null/empty.
    //   Emits "(not QR)" on DM and 1D symbologies.
    var _vibDebug = _qrBranch
        ? _enumKV22(_pick(q, "versionInformationBlock"), "q.versionInformationBlock")
        : "(not QR)";
    o += elem("DebugQVIBKeys", _vibDebug);

    // RETAINED BASELINE: DebugImagePolarity -- q.general 7-key inventory.
    //   Confirmed DM + QR (live + IMAGE.LOAD): verticalBWG/horizontalBWG/
    //   xDimension/contrastUniformity/contrastUniformityRow/contrastUniformityCol/MRD.
    //   No polarity/imagePolarity/eci/encodedCharacters. Retained as regression baseline.
    var _v30gnEnum = _enumKV22((q ? _pick(q, "general") : null), "q.general");
    var _v30polDesc = "gn.polarity="      + (q ? gnProp("polarity")      : "")
                    + ";gn.imagePolarity=" + (q ? gnProp("imagePolarity") : "")
                    + ";gn.image="         + (q ? gnProp("image")         : "")
                    + ";gn.colorMode="     + (q ? gnProp("colorMode")     : "")
                    + ";gn.imageColor="    + (q ? gnProp("imageColor")    : "")
                    + ";gn.eci="           + (q ? gnProp("eci")           : "")
                    + ";gn.ECI="           + (q ? gnProp("ECI")           : "")
                    + ";gn.eciMode="       + (q ? gnProp("eciMode")       : "")
                    + ";gn.encodedChars="  + (q ? gnProp("encodedCharacters") : "")
                    + ";allKeys=" + _v30gnEnum;
    o += elem("DebugImagePolarity", _v30polDesc);

    // DebugRImage -- DROPPED v1.33 (answered 2026-05-24 by scan #11):
    //   r.image = camera hardware metadata only (id/FoV/RoI/exposureTime/gain/
    //   autoExposure/illEnabled/focusLength/mirrorAngle/intIllWavelength/
    //   multicolorWavelength/multicolorRatio/multicolorName).
    //   NO ImagePolarity, ECI, ECLevel, DataMaskPattern. Path exhausted.


    o += '</DMSymVerResponse>\r\n'
       + '</DMCCResponse>';

    // Firmware 5.x / 6.x compatibility:
    // Older firmware exposes a global 'output' object; newer firmware passes
    // the output as the third parameter 'outputResults'.  Try both so the
    // script works across revisions without modification.
    if (typeof outputResults !== "undefined" && outputResults !== null) {
        outputResults.content = o;
    } else {
        output.content = o;
    }
